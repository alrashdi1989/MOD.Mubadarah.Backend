  GetUserReportDataSource = () => {
    return new CustomStore({
      key: 'id',
      load: async (loadOptions: LoadOptions<UserReportDto>) => {
        return await this.userReportService
          .getList(await loadOptionsStringify(loadOptions))
          .toPromise().then(result => {
            console.log('this.userReportService',this.userReportService)
            return result;
          }).catch(() => { throw 'LoadingError'; });
        },
        insert: (userId: string, values: string[]) => {
           return this.userReportService.create(userId,values).toPromise();
        },
    
      update: (key, values:UserReportDto) => {
        return this.userReportService.update(key, values).toPromise();
      },
      remove: key => {
    
        return this.userReportService.delete(key).toPromise();
      },
    });
  }
